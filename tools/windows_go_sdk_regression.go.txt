// Test fragment from Go commit f4ac29c3c743b30a1e1a2b7ef4eae2588ac5c6f3.
func TestNamedPipeConcurrentReadWrite(t *testing.T) {
	t.Parallel()
	name := pipeName()
	server := newBytePipe(t, name, true)
	client := newFileOverlapped(t, name, true)
	const count = 100
	var wg sync.WaitGroup
	for _, f := range []*os.File{server, client} {
		wg.Go(func() {
			var buf [1]byte
			for i := range count {
				if _, err := io.ReadFull(f, buf[:]); err != nil {
					t.Error(err)
					f.Close()
					return
				}
				if buf[0] != byte(i) {
					t.Errorf("Read = %d; want %d", buf[0], i)
				}
			}
		})
		wg.Go(func() {
			for i := range count {
				if _, err := f.Write([]byte{byte(i)}); err != nil {
					t.Error(err)
					f.Close()
					return
				}
			}
		})
	}
	wg.Wait()
}
